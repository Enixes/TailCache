```sh
python3 scripts/tailcache12_predictor_verify.py docs/predictor/predictor-data-20261118.json --out /tmp/predictor-check.json
python3 scripts/tailcache12_predictor_reproduce.py docs/predictor/predictor-data-20261118.json --solver docs/predictor/drain_candidate.py.txt --out /tmp/predictor-model-replay.json
```

```sh

python3 scripts/tailcache12_predictor_report.py
```
